using System.Collections.Concurrent; var builder = WebApplication.CreateBuilder(args); builder.Services.AddCors(options => { options.AddPolicy("AllowAll", policy => { policy.AllowAnyOrigin() .AllowAnyHeader() .AllowAnyMethod(); }); }); var app = builder.Build(); app.UseCors("AllowAll"); app.UseDefaultFiles(); app.UseStaticFiles(); var products = new List { new(1, "Milk 1L", "SKU-001", "Dairy", 45m, 52m, 10), new(2, "Bread", "SKU-002", "Bakery", 30m, 38m, 8), new(3, "Rice 5kg", "SKU-003", "Groceries", 350m, 420m, 18), new(4, "Laundry Detergent", "SKU-004", "Household", 220m, 260m, 12), new(5, "Sugar 1kg", "SKU-005", "Groceries", 80m, 95m, 14), new(6, "Soap", "SKU-006", "Personal Care", 55m, 68m, 20) }; var inventory = new Dictionary { [1] = 28, [2] = 41, [3] = 16, [4] = 9, [5] = 35, [6] = 24 }; var salesHistory = new List(); app.MapGet("/api/health", () => Results.Ok(new { status = "ok", timestamp = DateTime.UtcNow })); app.MapGet("/api/products", () => Results.Ok(products.Select(p => new { p.Id, p.Name, p.Sku, p.Category, p.CostPrice, p.SellingPrice, p.TaxRate, Quantity = inventory.GetValueOrDefault(p.Id, 0) }))); app.MapGet("/api/inventory", () => Results.Ok(inventory.Select(kvp => { var product = products.First(p => p.Id == kvp.Key); return new { product.Id, product.Name, Quantity = kvp.Value, product.SellingPrice, product.CostPrice, IsLowStock = kvp.Value <= product.ReorderLevel }; }))); app.MapGet("/api/dashboard", () => { var totalProducts = products.Count; var lowStockItems = inventory.Where(i => i.Value <= products.First(p => p.Id == i.Key).ReorderLevel).Count(); var totalStock = inventory.Values.Sum(); var weeklySales = salesHistory.Sum(s => s.TotalAmount); return Results.Ok(new { TotalProducts = totalProducts, LowStockItems = lowStockItems, TotalStock = totalStock, WeeklySales = weeklySales, TopCategories = new[] { new { Category = "Groceries", Total = 1650m }, new { Category = "Dairy", Total = 930m }, new { Category = "Household", Total = 840m } } }); }); app.MapGet("/api/reports/summary", () => { var summary = new { CashSales = 1850m, CardSales = 3240m, WalletSales = 680m, Returns = 190m, NetSales = 4820m, Alerts = new[] { "Milk stock below reorder level", "Rice stock under minimum threshold", "Laundry detergent is running low" } }; return Results.Ok(summary); }); app.MapPost("/api/sales/checkout", (CheckoutRequest request) => { var errors = new List(); foreach (var item in request.Items) { if (!inventory.TryGetValue(item.ProductId, out var stock) || stock < item.Quantity) { errors.Add($"Insufficient stock for product id {item.ProductId}."); } } if (errors.Count > 0) { return Results.BadRequest(new { errors }); } decimal subtotal = 0m; foreach (var item in request.Items) { var product = products.First(p => p.Id == item.ProductId); subtotal += product.SellingPrice * item.Quantity; } decimal tax = subtotal * 0.08m; decimal total = subtotal + tax; foreach (var item in request.Items) { inventory[item.ProductId] -= item.Quantity; } var sale = new SaleRecord { Id = Guid.NewGuid().ToString(), CustomerName = request.CustomerName ?? "Walk-in Customer", PaymentMethod = request.PaymentMethod, TotalAmount = total, Items = request.Items.Select(i => new SaleLineItem { ProductId = i.ProductId, Quantity = i.Quantity, UnitPrice = products.First(p => p.Id == i.ProductId).SellingPrice }).ToList(), CreatedAt = DateTime.UtcNow }; salesHistory.Add(sale); return Results.Ok(new { ReceiptNumber = $"POS-{sale.Id[..8].ToUpper()}", Subtotal = subtotal, Tax = tax, Total = total, PaymentMethod = request.PaymentMethod, Message = "Checkout processed successfully." }); }); app.MapFallbackToFile("index.html"); app.Run(); public record Product(int Id, string Name, string Sku, string Category, decimal CostPrice, decimal SellingPrice, int ReorderLevel); public class CheckoutRequest { public string? CustomerName { get; set; } public string PaymentMethod { get; set; } = "Cash"; public List Items { get; set; } = new(); } public class CartItemRequest { public int ProductId { get; set; } public int Quantity { get; set; } } public class SaleRecord { public string Id { get; set; } = string.Empty; public string CustomerName { get; set; } = string.Empty; public string PaymentMethod { get; set; } = string.Empty; public decimal TotalAmount { get; set; } public DateTime CreatedAt { get; set; } public List Items { get; set; } = new(); } public class SaleLineItem { public int ProductId { get; set; } public int Quantity { get; set; } public decimal UnitPrice { get; set; } }
